const ROOT_FOLDER_ID = "1dKd67sE0_wxMLXfzhCyQhRnXOcojVEjK"; const SECTION_KEYWORDS = { photos: ["фото", "хедшот", "photo", "headshot"], zsu: ["зсу", "волонтер", "pro bono", "zsu", "uaf"], stb: ["стб", "stb", "промо"], commercial: ["комерц", "commercial", "реклама"], social: ["соціальн", "social"] }; function doGet(e) { const section = (e && e.parameter && e.parameter.section) ? e.parameter.section.toLowerCase() : "photos"; const keywords = SECTION_KEYWORDS[section]; if (!keywords) { return createJsonResponse({ error: "Розділ не знайдено", items: [] }); } try { const rootFolder = DriveApp.getFolderById(ROOT_FOLDER_ID); const subFolders = rootFolder.getFolders(); let targetFolder = null; while (subFolders.hasNext()) { const folder = subFolders.next(); const folderName = folder.getName().toLowerCase(); // Щоб папка "Соціальна реклама" випадково не потрапила в "Комерційну" if (section === "commercial" && folderName.includes("соціальн")) { continue; } if (keywords.some(kw => folderName.includes(kw))) { targetFolder = folder; break; } } if (!targetFolder) { return createJsonResponse({ error: "Підпапку для розділу не знайдено: " + section, items: [] }); } const files = targetFolder.getFiles(); const items = []; while (files.hasNext()) { const file = files.next(); const mime = file.getMimeType(); const id = file.getId(); const name = file.getName().replace(/\.[^/.]+$/, ""); if (mime.startsWith("image/")) { items.push({ id: id, name: name, type: "image", thumb: "https://drive.google.com/thumbnail?id=" + id + "&sz=w800", full: "https://drive.google.com/thumbnail?id=" + id + "&sz=w2000" }); } else if (mime.startsWith("video/")) { items.push({ id: id, name: name, type: "video", thumb: "https://drive.google.com/thumbnail?id=" + id + "&sz=w800", streamUrl: "https://drive.google.com/uc?export=download&id=" + id, embedUrl: "https://drive.google.com/file/d/" + id + "/preview" }); } else if (mime.startsWith("audio/")) { items.push({ id: id, name: name, type: "audio", streamUrl: "https://drive.google.com/uc?export=download&id=" + id, embedUrl: "https://drive.google.com/file/d/" + id + "/preview" }); } } items.sort((a, b) => a.name.localeCompare(b.name, 'uk')); return createJsonResponse({ section: section, folderName: targetFolder.getName(), count: items.length, items: items }); } catch (err) { return createJsonResponse({ error: err.toString(), items: [] }); } } function createJsonResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
